# Ruokalogin järjestettävä taulukko ("Tilastot") Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new "Tilastot" page (via the Valikko sidebar) showing every logged food entry in a sortable, spreadsheet-style table across 19 real nutrient columns, with four time-granularity views (per-entry / per-day / per-week / per-month) and a matching CSV export.

**Architecture:** No database changes — every column this needs already exists on `food_cache`/`custom_foods` from the two prior Sulamo-comparison features (fiber/sugar/salt, then fat-split/minerals/vitamins). Pure additive UI: a new page fetches `food_log_entries` for a date range (default last 30 days) with the same join shape as `loadFoodDayEntries()`, computes 12 new `entryX()` value helpers (extending the existing `entryFiber()`/`entrySugar()`/`entrySalt()` pattern to the remaining nutrients), buckets/sorts entirely client-side (no refetch on granularity or sort change — data volume at the 30-day default is small), and renders a real `<table>` with a sticky first column for horizontal scroll. CSV export reuses the existing `downloadCSV()` helper, matching the pattern of `exportActivitiesCSV()`.

**Tech Stack:** Vanilla JS/CSS in a single static `index.html` file, Supabase Postgres/PostgREST. No build step, no test framework — verification is manual.

**Spec:** `docs/superpowers/specs/2026-08-24-tilastot-design.md`

function tilastotWeekKey(dateIso) {
  const d = new Date(dateIso);
  return `${isoWeekYear(d)}-W${String(isoWeek(d)).padStart(2, '0')}`;
}

function tilastotBucketKey(entry, granularity) {
  if (granularity === 'day') return entry.logged_at;
  if (granularity === 'week') return tilastotWeekKey(entry.logged_at);
  return entry.logged_at.slice(0, 7);
}

function tilastotBuildRows() {
  if (tilastotGranularity === 'entry') {
    return tilastotEntries.map(entry => ({
      label: entry.logged_at,
      meal: MEAL_DEFS.find(m => m.key === entry.meal_type)?.label || entry.meal_type,
      name: foodItemName(entry),
      count: null,
      values: TILASTOT_COLUMNS.map(col => col.value(entry)),
    }));
  }
  const buckets = new Map();
  tilastotEntries.forEach(entry => {
    const key = tilastotBucketKey(entry, tilastotGranularity);
    if (!buckets.has(key)) buckets.set(key, []);
    buckets.get(key).push(entry);
  });
  return [...buckets.entries()].map(([key, entries]) => ({
    label: key,
    meal: null,
    name: null,
    count: entries.length,
    values: TILASTOT_COLUMNS.map(col => entries.reduce((sum, e) => sum + col.value(e), 0)),
  }));
}

function tilastotSortValue(row, key) {
  if (key === 'label') return row.label;
  if (key === 'meal') return row.meal || '';
  if (key === 'name') return row.name || '';
  if (key === 'count') return row.count ?? 0;
  const idx = TILASTOT_COLUMNS.findIndex(c => c.key === key);
  return row.values[idx];
}

function tilastotSortRows(rows) {
  const sorted = [...rows].sort((a, b) => {
    const av = tilastotSortValue(a, tilastotSortKey);
    const bv = tilastotSortValue(b, tilastotSortKey);
    if (typeof av === 'string' || typeof bv === 'string') {
      return String(av).localeCompare(String(bv));
    }
    return av - bv;
  });
  if (tilastotSortDir === 'desc') sorted.reverse();
  return sorted;
}

function sortTilastotBy(key) {
  if (tilastotSortKey === key) {
    tilastotSortDir = tilastotSortDir === 'asc' ? 'desc' : 'asc';
  } else {
    tilastotSortKey = key;
    tilastotSortDir = 'asc';
  }
  renderTilastotTable();
}
